        function updateSquadRadar() {

            const config = LEAGUE_CONFIG[activeLeagueKey];

            const starters = getActiveSportPlayers();

            if (starters.length === 0) return;


            const aggregates = [0, 0, 0, 0, 0, 0];

            starters.forEach(item => {

                item.player.radar.forEach((val, idx) => aggregates[idx] += val);

            });

            const avgData = aggregates.map(v => Math.round(v / starters.length));


            const ctx = document.getElementById('squadRadarChart').getContext('2d');

            if (squadRadarChart) squadRadarChart.destroy();


            const datasets = [{

                borderWidth: 2

            }];

        function poissonRandom(lambda) {

            let L = Math.exp(-lambda), k = 0, p = 1;

            do { k++; p *= Math.random(); } while (p > L);

            return k - 1;

        }


        function normalRandom(mean, std) {

            let u = 0, v = 0;

            while(u === 0) u = Math.random();

            while(v === 0) v = Math.random();

            return mean + std * Math.sqrt(-2.0 * Math.log(u)) * Math.cos(2.0 * Math.PI * v);

        }
